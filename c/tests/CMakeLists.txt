cmake_minimum_required(VERSION 3.18)
project(doc2struct_scan_tests LANGUAGES C)
option(DOC2STRUCT_SCALAR_SCAN "Disable custom SIMD scans" OFF)
enable_testing()
add_executable(doc2struct_scan_test test_scan.c)
target_include_directories(doc2struct_scan_test PRIVATE ../src)
target_compile_features(doc2struct_scan_test PRIVATE c_std_11)
if(DOC2STRUCT_SCALAR_SCAN)
    target_compile_definitions(doc2struct_scan_test PRIVATE DS_SCALAR_SCAN=1)
endif()
add_test(NAME bounded_scans COMMAND doc2struct_scan_test)
